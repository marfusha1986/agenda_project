
-- 1. Ana Görevler Tablosu
CREATE TABLE tasks (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR2(255) NOT NULL,
    due_date VARCHAR2(50),
    is_recurring NUMBER(1) DEFAULT 0,
    status VARCHAR2(50) DEFAULT 'pending',
    postpone_count NUMBER DEFAULT 0,
    category VARCHAR2(100)
);

-- 2. Görev Geçmişi / Loglar Tablosu
CREATE TABLE task_logs (
    id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    task_id NUMBER,
    action_type VARCHAR2(100),
    log_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_task FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE
);

-- Mevcut tabloyu güncellerken due_date kolonunu DATE tipine çevirelim
ALTER TABLE tasks MODIFY due_date DATE;

-- Tablodaki tüm verileri temizleyip ID'yi 1'den başlatmak için:
TRUNCATE TABLE tasks;
